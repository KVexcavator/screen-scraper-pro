могу использовать в коде под винду
```rust
use std::process::Command;
use std::path::Path;

fn convert_video_to_rgba(input: &str, output: &str) -> anyhow::Result<()> {
    let ffmpeg_path = "./bin/ffmpeg.exe";

    if !Path::new(ffmpeg_path).exists() {
        anyhow::bail!("ffmpeg.exe не найден по пути {}", ffmpeg_path);
    }

    let status = Command::new(ffmpeg_path)
        .args(&["-i", input, "-vf", "format=rgba", output])
        .status()?;

    if !status.success() {
        anyhow::bail!("ffmpeg завершился с ошибкой");
    }

    Ok(())
}

fn main() -> anyhow::Result<()> {
    convert_video_to_rgba("input.mp4", "output_rgba.mp4")?;
    println!("Видео успешно конвертировано в RGBA!");
    Ok(())
}
```
