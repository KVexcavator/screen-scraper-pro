---
простой FrameBus на Rust
```rust
pub struct FrameBus {
    subscribers: Vec<Sender<Vec<u8>>>,
}

impl FrameBus {
    pub fn new() -> Self {
        Self {
            subscribers: Vec::new()
        }
    }

    pub fn subscribe(&mut self) -> Receiver<Vec<u8>> {
        let (tx, rx) = std::sync::mpsc::channel();
        self.subscribers.push(tx);
        rx
    }

    pub fn publish(&mut self, frame: Vec<u8>) {
        for sub in &self.subscribers {
            let _ = sub.send(frame.clone());
        }
    }
}
```
CaptureEngine тогда выглядит так
```rust

    let frame = capture_frame();
    frame_bus.publish(frame);
}
```

Правильный тип кадра Arc<Vec<u8>> или Arc<[u8]>
Тогда publish
```rust
pub fn publish(&mut self, frame: Arc<Vec<u8>>) {
    for sub in &self.subscribers {
        let _ = sub.send(frame.clone());
    }
}
```
clone тут не копирует память
