## Шаг 3 — Пайп в ffmpeg
```rust
use std::process::{Command, Stdio};
use std::io::Write;

let mut child = Command::new("ffmpeg")
    .args([...])
    .stdin(Stdio::piped())
    .spawn()
    .unwrap();

let stdin = child.stdin.as_mut().unwrap();

stdin.write_all(frame_bytes)?;
```
